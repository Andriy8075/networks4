# Мережевий простір проксі: запис DNS, TCP та HTTP
# (через скрипт lab4/code/02_capture_normal.sh або вручну:)

sudo nsenter --target "$PROXY_PID" --net -- tcpdump -i any -n -U -w /tmp/normal.pcap

# зупинити tcpdump (Ctrl+C), скопіювати PCAP у lab4/evidence/pcaps/normal.pcap
